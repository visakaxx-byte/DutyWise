def test_full_workflow():
    # 1. 上传文件
    response = client.post('/api/v1/shipments/upload', files=files)
    shipment_id = response.json()['data']['shipment_id']

    # 2. 开始处理
    response = client.post(f'/api/v1/shipments/{shipment_id}/process')
    task_id = response.json()['data']['task_id']

    # 3. 等待完成
    while True:
        response = client.get(f'/api/v1/tasks/{task_id}/status')
        if response.json()['data']['status'] == 'completed':
            break
        time.sleep(1)

    # 4. 验证结果
    response = client.get(f'/api/v1/shipments/{shipment_id}/result')
    assert response.json()['data']['status'] == 'completed'

    start_time = time.time()

    # 处理一票数据（100个商品）
    process_shipment(shipment_id)

    elapsed_time = time.time() - start_time

    # 应该在2分钟内完成
    assert elapsed_time < 120

def test_concurrent_processing():
    """测试并发处理"""
    # 同时处理10票数据
    tasks = []
    for i in range(10):
        task = process_shipment_task.delay(shipment_ids[i])
        tasks.append(task)

    # 等待所有任务完成
    for task in tasks:
        task.get(timeout=300)

    # 验证所有任务都成功
    assert all(task.successful() for task in tasks)

## 5.6 风险管理

### 5.6.1 技术风险

| 风险 | 概率 | 影响 | 应对措施 |
|------|------|------|---------|
| AI识别准确率低 | 中 | 高 | 提供人工校验功能 |
