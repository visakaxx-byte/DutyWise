## 代码实现

### 字段映射（使用豆包）

```python
import requests
import json
from typing import List, Dict

class DoubaoAI:
    """豆包AI封装"""

    def __init__(self, api_key: str):
        self.api_key = api_key
        self.base_url = "https://ark.cn-beijing.volces.com/api/v3"

    def chat(self, messages: List[Dict], model: str = "ep-20240611125520-lmk7v") -> str:
        """调用豆包对话接口"""
        response = requests.post(
            f"{self.base_url}/chat/completions",
            headers={
                "Authorization": f"Bearer {self.api_key}",
                "Content-Type": "application/json"
            },
            json={
                "model": model,
                "messages": messages,
                "temperature": 0.1  # 降低随机性，提高准确性
            }
        )

        if response.status_code != 200:
            raise Exception(f"豆包API调用失败: {response.text}")

        result = response.json()
        return result['choices'][0]['message']['content']


class AIFieldMapper:
    """AI字段映射器"""

    def __init__(self, doubao_api_key: str):
        self.ai = DoubaoAI(doubao_api_key)

    def map_fields(self, source_fields: List[str], target_fields: List[str]) -> Dict:
        """使用AI进行字段映射"""

        prompt = f"""
你是一个专业的数据映射助手。

源字段列表：
{json.dumps(source_fields, ensure_ascii=False, indent=2)}

目标字段列表：
{json.dumps(target_fields, ensure_ascii=False, indent=2)}

请将源字段映射到目标字段，返回JSON格式：
{{
    "mappings": [
        {{"source": "源字段名", "target": "目标字段名", "confidence": 0.95}},
        ...
    ]
}}

规则：
1. 只映射有明确对应关系的字段
2. confidence 表示置信度（0-1）
3. 如果源字段无法映射，不要包含在结果中
"""

        messages = [{"role": "user", "content": prompt}]
        response = self.ai.chat(messages)

        # 解析JSON
        try:
            return json.loads(response)
        except json.JSONDecodeError:
            # 如果返回的不是纯JSON，尝试提取
            import re
            json_match = re.search(r'\{.*\}', response, re.DOTALL)
            if json_match:
                return json.loads(json_match.group())
            raise Exception("AI返回格式错误")


# 使用示例
mapper = AIFieldMapper(api_key="your_doubao_api_key")

source_fields = ["品名", "货物名称", "Description", "数量", "QTY", "毛重"]
target_fields = ["中文品名", "英文品名", "数量", "单位", "净重", "毛重"]

result = mapper.map_fields(source_fields, target_fields)
print(result)

# 输出示例：
# {
#     "mappings": [
#         {"source": "品名", "target": "中文品名", "confidence": 0.95},
#         {"source": "货物名称", "target": "中文品名", "confidence": 0.90},
#         {"source": "Description", "target": "英文品名", "confidence": 0.98},
#         {"source": "数量", "target": "数量", "confidence": 1.0},
#         {"source": "QTY", "target": "数量", "confidence": 0.95},
#         {"source": "毛重", "target": "毛重", "confidence": 1.0}
#     ]
# }
```

## 品类相似度判断（使用豆包）

```python
class HSSimilarityChecker:
    """HS编码相似度检查器"""

    def __init__(self, doubao_api_key: str):
        self.ai = DoubaoAI(doubao_api_key)

    def check_similarity(self, original_item: Dict, candidate_item: Dict) -> Dict:
        """检查两个商品的相似度"""

        prompt = f"""
你是一个海关商品分类专家。

原始商品：
- 品名：{original_item.get('中文品名')}
- 材质：{original_item.get('材质')}
- 用途：{original_item.get('用途')}
- HS编码：{original_item.get('商品编码')}

候选商品：
- 品名：{candidate_item.get('description_cn')}
- HS编码：{candidate_item.get('hs_code_cn')}

请判断这两个商品是否属于相似品类，可以互相替换用于报关。

返回JSON格式：
{{
    "is_similar": true/false,
    "similarity_score": 0.85,  // 0-1之间
    "reason": "相似原因说明",
    "risk_level": "LOW/MEDIUM/HIGH",
    "warnings": ["可能的风险提示"]
}}

判断标准：
1. 品类是否接近（如：阀门配件 vs 阀门垫圈件）
2. 材质是否一致（如：不锈钢 vs 钢铁）
3. 用途是否相同（如：工业用 vs 家用）
4. HS编码前缀是否相同
"""

        messages = [{"role": "user", "content": prompt}]
        response = self.ai.chat(messages)

        return json.loads(response)


# 使用示例
checker = HSSimilarityChecker(api_key="your_doubao_api_key")

original = {
    "中文品名": "阀门配件",
    "材质": "不锈钢",
    "用途": "工业",
    "商品编码": "8481901000"
}

candidate = {
    "description_cn": "阀门垫圈件",
    "hs_code_cn": "8481900085"
}

result = checker.check_similarity(original, candidate)
print(result)
```

### 2. 缓存结果

```python
import hashlib

def get_cache_key(original: Dict, candidate: Dict) -> str:
    """生成缓存key"""
    content = f"{original['中文品名']}_{candidate['description_cn']}"
    return hashlib.md5(content.encode()).hexdigest()

def check_similarity_with_cache(self, original: Dict, candidate: Dict) -> Dict:
    """带缓存的相似度检查"""
    cache_key = get_cache_key(original, candidate)

    # 先查缓存
    cached = redis_client.get(f"similarity:{cache_key}")
    if cached:
        return json.loads(cached)

    # 调用AI
    result = self.check_similarity(original, candidate)

    # 写入缓存（7天）
    redis_client.setex(
        f"similarity:{cache_key}",
        7 * 24 * 3600,
        json.dumps(result)
    )

    return result
```
